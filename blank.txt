import { getGalleryItems } from "@/app/lib/api/gallery"; // Adjust path to your function if needed
import { NextResponse } from "next/server";

export async function GET(request) {
    try {
        // Extract URL search parameters (e.g., /api/gallery?page=1&limit=6)
        const { searchParams } = new URL(request.url);
        
        const page = parseInt(searchParams.get("page")) || 1;
        const limit = parseInt(searchParams.get("limit")) || 10;
        const category = searchParams.get("category"); // Optional category filter if supported

        // Call your existing paginated function
        const result = await getGalleryItems({ page, limit, category });

        return NextResponse.json(
            { 
                success: true,
                data: result 
            },
            { status: 200 }
        );
    } catch (error) {
        console.error("Error fetching gallery items:", error);
        return NextResponse.json(
            { success: false, error: "Internal server error !!!" },
            { status: 500 }
        );
    }
}
